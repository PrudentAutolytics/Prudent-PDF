Quickstart

Submit a PDF for redaction. This is a preview of the planned request format and may change before release.

# Submit a PDF for redaction
curl -X POST https://redact.prudentautolytics.com/api/jobs-submit \
  -H "Authorization: Bearer $PRUDENT_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "fileName": "contract.pdf",
    "fileBase64": "<base64 encoded PDF>"
  }'
# Submit a PDF for redaction
$bytes = [IO.File]::ReadAllBytes("contract.pdf")
$body  = @{ fileName = "contract.pdf"; fileBase64 = [Convert]::ToBase64String($bytes) } | ConvertTo-Json
Invoke-RestMethod -Method Post -Uri "https://redact.prudentautolytics.com/api/jobs-submit" `
  -Headers @{ Authorization = "Bearer $env:PRUDENT_API_KEY" } `
  -ContentType "application/json" -Body $body
# Submit a PDF for redaction
import base64, os, requests

with open("contract.pdf", "rb") as f:
    pdf = base64.b64encode(f.read()).decode()

r = requests.post(
    "https://redact.prudentautolytics.com/api/jobs-submit",
    headers={"Authorization": f"Bearer {os.environ['PRUDENT_API_KEY']}"},
    json={"fileName": "contract.pdf", "fileBase64": pdf},
)
print(r.json())
